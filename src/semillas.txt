# Semillas a mano: 'pNN 0xADDR b6000 b8000 bA000  # por que'.
#
# El motor de sonido (p13:6000, lo llama la interrupcion con 13-14-15 en
# p00:4056) salta a cada canal con `jp (hl)` (p13:6073) por el puntero que
# hay en 0xC010/0xC012/0xC014/0xC016. Esos punteros los escriben:
p13 0x6074 13 14 15  # p00:56D3 y p13:602A: ld hl,0x6074 / ld (0xC010..6),hl (canal sin nada que tocar)
p13 0x6095 13 14 15  # a mano: p13:608B pone 0x6095 en el puntero del canal (es un `ret`: canal parado)
p13 0x6134 13 14 15  # p00:5794: ld hl,0x6134 / ld (0xC010),hl
p13 0x613B 13 14 15  # p00:579A: ld hl,0x613B / ld (0xC012),hl
p13 0x6142 13 14 15  # p00:57A0: ld hl,0x6142 / ld (0xC014),hl
p13 0x64D7 13 14 15  # p00:5826: ld hl,0x64D7 / ld (0xC010),hl
p13 0x64DD 13 14 15  # p00:582C: ld hl,0x64DD / ld (0xC012),hl
p13 0x64E3 13 14 15  # p00:5832: ld hl,0x64E3 / ld (0xC014),hl
p13 0x64E9 13 14 15  # p00:587C: ld hl,0x64E9 / ld (0xC016),hl
p13 0x64D1 13 14 15  # p00:58A3: ld hl,0x64D1 / ld (0xC127),hl; p13:60B4 lo salta con `jp (hl)` (sonidos 0x70-0x7F)
p13 0x6126 13 14 15  # p00:5964: ld hl,0x6126 / ld (0xC018),hl; p13:6099 lo salta (sonido 0xFD)
p13 0x612D 13 14 15  # p00:596A: ld hl,0x612D / ld (0xC01A),hl; p13:60A3 lo salta (sonido 0xFD)
p00 0x461D 1 2 3  # a mano: p00:40ED pone 0x461D en la pila (ld hl / push hl) antes de despachar los estados 0-2: es su vuelta
#
# CODIGO SIN ENTRADA. Rutinas enteras (acaban en `ret` o caen en codigo ya
# trazado) a las que ninguna instruccion del cartucho salta ni apunta con un
# inmediato de 16 bits (tools/huecos.py, 2026-10-12). Van como codigo para que
# el listado las lea; NO esta medido que no se ejecuten por un camino
# calculado.
